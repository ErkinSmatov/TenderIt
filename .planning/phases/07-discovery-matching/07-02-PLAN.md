LAST_POLLED_KEY = "discovery:last_polled_at"
DEFAULT_LOOKBACK_DAYS = 7

async def poll_goszakup_discovery(ctx: dict) -> None:
  """ARQ cron: batch-fetch new/updated tenders from goszakup and upsert to DB.
  Called every 15 min by WorkerSettings.cron_jobs (unique=True prevents overlap).
  """
  redis = ctx["redis"]

  # 1. Read last_polled_at from Redis (default to 7 days ago on first run)
  ts = await redis.get(LAST_POLLED_KEY)
  since = datetime.fromisoformat(ts.decode()) if ts else (utcnow() - timedelta(days=DEFAULT_LOOKBACK_DAYS))

  # 2. Paginated batch fetch with T-07-04 inter-page delay
  all_tender_dicts = []
  offset = 0
  LIMIT = 50
  while True:
    batch = await fetch_tenders_batch(since, limit=LIMIT, offset=offset)
    if not batch:
      break
    all_tender_dicts.extend(batch)
    if len(batch) < LIMIT:
      break
    offset += LIMIT
    await asyncio.sleep(0.5)  # T-07-04: conservative rate — goszakup has no public SLA

  if not all_tender_dicts:
    logger.info("poll_goszakup_discovery: no new/updated tenders since %s", since)
    await redis.set(LAST_POLLED_KEY, utcnow().isoformat())
    return

  # 3. Upsert each tender batch to DB
  async with ctx["db_session_factory"]() as session:
    upserted_ids = await _upsert_tenders(session, all_tender_dicts)

  # 4. Write last_polled_at to Redis (only after successful upsert)
  await redis.set(LAST_POLLED_KEY, utcnow().isoformat())
  logger.info("poll_goszakup_discovery: upserted %d tenders, enqueuing run_matching", len(upserted_ids))

  # 5. Enqueue run_matching with list of upserted tender IDs
  if upserted_ids:
    await redis.enqueue_job("run_matching", upserted_ids)
