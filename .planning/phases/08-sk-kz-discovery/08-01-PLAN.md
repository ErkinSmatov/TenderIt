Output: sk_kz_service.py (stateless REST client) and poll_sk_kz_discovery.py (stateful ARQ cron task). Registration in worker_settings.py is intentionally deferred to Plan 08-03 (Wave 2) so it can be tested first.

1. async def poll_sk_kz_discovery(ctx: dict) -> None
   Docstring: "ARQ cron: batch-fetch new/updated tenders from zakup.sk.kz and upsert to DB."
   
   Step 1 — Compute since:
     raw_ts = await ctx["redis"].get(LAST_POLLED_KEY)
     if raw_ts:
         since = datetime.fromisoformat(raw_ts)  (raw_ts is a string — fakeredis decode_responses=True)
     else:
         since = datetime.now(tz=timezone.utc) - timedelta(hours=DEFAULT_LOOKBACK_HOURS)
     logger.info("poll_sk_kz_discovery: polling since %s", since.isoformat())
   
   Step 2 — Fetch (single page; 15-min interval with 24h lookback means <50 new tenders):
     all_tender_dicts = await fetch_sk_tenders_page(since, page=0, size=_PAGE_SIZE)
   
   Step 3 — Early exit if no results:
     if not all_tender_dicts:
         logger.info("poll_sk_kz_discovery: no new/updated tenders since %s", since.isoformat())
         await ctx["redis"].set(LAST_POLLED_KEY, datetime.now(tz=timezone.utc).isoformat())
         return
   
   Step 4 — Upsert (ONLY inside the context manager):
     async with ctx["db_session_factory"]() as session:
         upserted_ids = await _upsert_tenders(session, all_tender_dicts)
   
   Step 5 — Write Redis timestamp ONLY after successful upsert (atomicity invariant):
     await ctx["redis"].set(LAST_POLLED_KEY, datetime.now(tz=timezone.utc).isoformat())
     logger.info("poll_sk_kz_discovery: upserted %d tenders", len(upserted_ids))
   
   Step 6 — Enqueue matching:
     if upserted_ids:
         await ctx["redis"].enqueue_job("run_matching", upserted_ids)
